double get overallProgress {
  if (totalRequests == 0) return 1.0;
  final progress = doneRequests / totalRequests;
  // debugPrint('Fortschritt: $doneRequests von $totalRequests Requests = ${(progress * 100).toStringAsFixed(1)}%');
  return progress;
}
